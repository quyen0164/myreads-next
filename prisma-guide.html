Toàn Tập Dự Án MyReads (Next.js + Prisma)

Tài liệu này chứa toàn bộ mã nguồn và giải thích chi tiết cơ chế hoạt động (OOP, Memory, Async/Await).

1. Lệnh Terminal (Cài đặt & Khởi tạo)

# 1. Tạo dự án Next.js. Lệnh này cấp phát thư mục, cài React, TypeScript, Tailwind.
npx create-next-app@latest myreads-next
cd myreads-next

# 2. Cài Prisma 7 (Bản ổn định hỗ trợ SQLite).
# prisma (CLI tool) cài vào devDependencies để tạo bảng.
npm install prisma@7 --save-dev
# @prisma/client (ORM engine) cài vào dependencies để chạy app.
npm install @prisma/client@7

# 3. Cài Adapter cho SQLite (Prisma 7 bắt buộc phải có)
# better-sqlite3 là thư viện C++ giao tiếp vật lý với file .db
npm install @prisma/adapter-better-sqlite3 better-sqlite3

# 4. Cài tsx để chạy file seed.ts (TypeScript) trực tiếp bằng Node.js
npm install -D tsx

# 5. Khởi tạo Prisma. Tạo folder prisma/ và file schema.prisma.
npx prisma@prev init --datasource-provider sqlite

2. File `.env` (Môi trường)

# Chứa Connection String trỏ đến file SQLite cục bộ.
# `file:` là protocol, `./` là thư mục hiện tại.
DATABASE_URL="file:./dev.db"

3. File `prisma7.config.ts` (Cấu hình CLI)

import 'dotenv/config'; // Load file .env vào process.env
import { defineConfig } from 'prisma/config';

export default defineConfig({
  // Chỉ định file schema chứa các Model.
  schema: 'prisma/schema.prisma',
  
  migrations: {
    // Lệnh dùng khi chạy `npx prisma db seed`.
    // Dùng tsx để thực thi file TS thay vì Node thuần.
    seed: 'tsx ./prisma/seed.ts', 
  },
  
  datasource: {
    // Lấy url từ biến môi trường. Dấu ! khẳng định biến chắc chắn có giá trị (không undefined).
    url: process.env.DATABASE_URL!,
  },
});

4. File `prisma/schema.prisma` (Định nghĩa Database)

// Cấu hình Prisma Client sinh mã nguồn TypeScript.
generator client {
  provider = "prisma-client-js"
  output   = "../generated/prisma" // Nơi chứa code sinh ra (import từ đây)
}

datasource db {
  provider = "sqlite"
}

// Model User: Định nghĩa bảng Users
model User {
  id    String @id @default(uuid()) 
  // @id: Primary Key. @default(uuid()): Hàm sinh chuỗi ID ngẫu nhiên (VD: 550e8400).
  
  email String @unique              
  // @unique: Tạo UNIQUE INDEX. Nếu insert trùng email, DB báo lỗi.
  
  name  String
  
  books Book[]                        
  // [Quan hệ 1-N]: Trường ảo, KHÔNG tạo cột trong SQL. Dùng để Prisma query: user.books
}

// Model Book: Định nghĩa bảng Books
model Book {
  id     String @id @default(uuid())
  title  String
  shelf  String @default("none")
  
  userId String                   
  // [Foreign Key]: Tạo cột thật trong SQLite, lưu ID của User.
  
  user   User   @relation(fields: [userId], references: [id], onDelete: Cascade)
  // @relation: Khai báo Foreign Key Constraint.
  // onDelete: Cascade: Cơ chế Trigger. Xóa User cha -> Tự động xóa Sách của user đó.
}

5. File `prisma/seed.ts` (Bơm dữ liệu mẫu)

import { PrismaClient } from '../generated/prisma';
import { PrismaBetterSqlite3 } from '@prisma/adapter-better-sqlite3';

// 1. Khởi tạo Driver (Adapter). `new` cấp phát RAM cho instance.
// Truyền url để Adapter biết file DB vật lý nằm ở đâu.
const adapter = new PrismaBetterSqlite3({ 
  url: 'file:./dev.db' 
});

// 2. Khởi tạo ORM Engine và Tiêm (Dependency Injection) Adapter vào.
const prisma = new PrismaClient({ adapter });

async function main() {
  // 3. `await` tạm dừng luồng, chờ lệnh INSERT chạy xong.
  const user = await prisma.user.create({
    data: { name: "Admin", email: "admin@test.com" }
  });

  // 4. Sử dụng ID của user vừa tạo để gán Foreign Key cho sách.
  await prisma.book.create({
    data: { title: "Clean Code", shelf: "read", userId: user.id }
  });
}

main().catch(e => console.error(e));

6. File `lib/prisma.ts` (Kết nối toàn cục)

Tại sao cần file này? Next.js có tính năng Hot Reload (sửa code là tải lại ngay). Nếu cứ tạo `new PrismaClient()` mỗi lần tải lại, hàng trăm kết nối DB sẽ bị rò rỉ (Memory Leak). Ta dùng biến `global` để tái sử dụng duy nhất 1 kết nối. 
import { PrismaClient } from '../generated/prisma';
import { PrismaBetterSqlite3 } from '@prisma/adapter-better-sqlite3';

const adapter = new PrismaBetterSqlite3({ url: 'file:./dev.db' });

// Khai báo type cho biến global để TypeScript không báo lỗi.
const globalForPrisma = global as unknown as { prisma: PrismaClient };

// Logic: Dùng lại (||) instance cũ trong global, nếu chưa có mới tạo mới.
export const prisma = globalForPrisma.prisma || new PrismaClient({ adapter });

if (process.env.NODE_ENV !== 'production') globalForPrisma.prisma = prisma;

7. File `app/page.tsx` (Server Component)

import { prisma } from '../lib/prisma';
import { updateShelf } from '../actions/books';

// `async` function: Code chạy trên Server. Next.js hỗ trợ Component bất đồng bộ.
export default async function Home() {
  
  // `await`: Tạm dừng luồng, nhường CPU cho request khác trong khi chờ I/O đọc file DB.
  // `.findMany()`: Gọi ORM sinh câu lệnh SELECT * FROM Book.
  // `include`: Yêu cầu thực hiện JOIN hoặc query phụ để lấy User liên quan.
  const books = await prisma.book.findMany({
    include: { user: true }
  });

  return (
    <main style={{ padding: "40px" }}>
      <h1>Tủ Sách</h1>
      <ul>
        {/* `.map()`: Trả về mảng React Elements. `key` giúp thuật toán Diff của React tối ưu DOM. */}
        {books.map((book) => (
          <li key={book.id}>
            <strong>{book.title}</strong> - {book.user.name}
            
            {/* Form gọi Server Action trực tiếp (không qua API Route) */}
            <form action={async (formData) => {
              "use server";
              const newShelf = formData.get("shelf") as string;
              await updateShelf(book.id, newShelf);
            }}>
              <select name="shelf" defaultValue={book.shelf} onChange={(e) => e.target.form?.requestSubmit()}>
                <option value="read">Đã đọc</option>
                <option value="none">Bỏ chọn</option>
              </select>
            </form>
          </li>
        ))}
      </ul>
    </main>
  );
}

8. File `actions/books.ts` (Server Action)

"use server"; // Directive: Báo Next.js Compiler đây là Server Code, tạo RPC Endpoint.

import { prisma } from '../lib/prisma';
import { revalidatePath } from 'next/cache';

export async function updateShelf(bookId: string, shelf: string) {
  // 1. Thực thi lệnh UPDATE vào Database thông qua ORM.
  await prisma.book.update({
    where: { id: bookId }, // Mệnh đề WHERE
    data: { shelf: shelf } // Mệnh đề SET
  });

  // 2. Cache Invalidation.
  // Next.js lưu cache HTML của trang để load cực nhanh. 
  // Hàm này xóa cache của route '/', buộc Next.js phải chạy lại Component để lấy data mới.
  revalidatePath('/');
}
